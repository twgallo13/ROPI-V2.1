Then create a new component `VocabularyValuesList.tsx`:

```tsx
import { useEffect, useState } from 'react';
import { doc, getDoc } from 'firebase/firestore';
import { db } from '../../../config/firebase';

export function VocabularyValuesList({ listPath }: { listPath: string }) {
  const [values, setValues] = useState<string[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchVocab() {
      try {
        // Parse path like "settings/lists/departments"
        const pathParts = listPath.split('/');
        if (pathParts.length < 2) throw new Error('Invalid path format');
        
        const docRef = doc(db, ...pathParts);
        const snapshot = await getDoc(docRef);
        
        if (snapshot.exists()) {
          const data = snapshot.data();
          setValues(data.values || []);
        } else {
          setError('List not found in Firestore');
        }
      } catch (err: any) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    }
    
    fetchVocab();
  }, [listPath]);

  if (loading) return <p className="text-sm text-gray-500">Loading vocabulary...</p>;
  if (error) return <p className="text-sm text-red-600">⚠️ {error}</p>;
  if (values.length === 0) return <p className="text-sm text-gray-500">No values found</p>;

  return (
    <div className="mt-2 p-2 bg-gray-50 rounded">
      <p className="text-xs font-medium text-gray-700 mb-1">Vocabulary Values ({values.length}):</p>
      <div className="flex flex-wrap gap-1">
        {values.slice(0, 20).map((val, i) => (
          <span key={i} className="px-2 py-0.5 bg-blue-100 text-blue-700 text-xs rounded">
            {val}
          </span>
        ))}
        {values.length > 20 && <span className="text-xs text-gray-500">+{values.length - 20} more</span>}
      </div>
    </div>
  );
}
```
